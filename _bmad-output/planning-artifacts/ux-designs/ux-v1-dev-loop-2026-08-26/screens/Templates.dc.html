# initiative · plan every epic in epics.yaml, then every story
name: initiative
version: 1
inputs:
project: { type: project }
repository: { type: repository }
base_branch: { type: string, default: main }
initiative_folder: { type: path }
harness: { type: enum, values: [claude, codex, local] }
budget_usd: { type: number, default: 40 }
re_review: { type: enum, values: [off, on], default: off }
review_plan: { type: enum, values: [off, on], default: off }
steps:
- id: plan
uses: plan.check
with: { file: "{{initiative_folder}}/epics.yaml" }
gate: deterministic
- id: review
uses: human
when: "{{review_plan}} == on"
gate: human
- id: epics
for_each: "{{plan.epics}}"
as: epic
uses: template
with:
name: epic
version: 1
working_branch: "{{epic.branch}}"
spec_folder: "{{epic.spec_folder}}"
harness: "{{harness}}"
re_review: "{{re_review}}"
- id: summary
run: git log --oneline "{{base_branch}}..HEAD"
# no gate → unvalidated
